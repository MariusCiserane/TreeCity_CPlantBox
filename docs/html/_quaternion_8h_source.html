308  // This Method calculates the Quaternion that rotates a to b
309  // this is basically the same as the implementation in the Unreal Engine
310  inline static Quaternion geodesicRotation(Vector3d a, Vector3d b)
311  {
312  double w = 1.0;
313  // normalize the vectors
314  a.normalize();
315  b.normalize();
316  // if the vectors are parallel, the rotation axis is undefined
317  if(w < std::numeric_limits<double>::epsilon() && std::abs(a.x) <= std::abs(a.z))
318  return Quaternion(w, Vector3d(0, -a.z, a.y));
319  // case 2 of the above
320  else if(w < std::numeric_limits<double>::epsilon() && std::abs(a.x) > std::abs(a.z))
321  return Quaternion(w, Vector3d(-a.y, a.x, 0));
322  // otherwise, the rotation axis is the cross product of a and b
323  else
324  return Quaternion(w, a.cross(b)).normalized();
325  }

328  // this method provides a spherical interpolation between two quaternions
329  inline static Quaternion SphericalInterpolation(Quaternion a, Quaternion b, double t = 0.5)
330  {
331  double angle = a.dot(b);
332  if (angle >= 1.0)
333  {
334  return a;
335  }
336  double half = acos(angle);
337  double sinHalf = sin(half);
338  if(std::abs(sinHalf) < std::numeric_limits<double>::epsilon()) //< if angle is 0.0, the quaternions are opposite
339  {
340  return {0.5*a.w + 0.5*b.w, 0.5*a.v + 0.5*b.v};
341  }
342  //< otherwise, interpolate
343  double ratioA = sin((1.0 - t) * half) / sinHalf;
344  double ratioB = sin(t * half) / sinHalf;
345  return {ratioA * a.w + ratioB * b.w, ratioA * a.v + ratioB * b.v};
346  }
