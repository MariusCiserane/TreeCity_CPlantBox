98 double SDF_RootSystem::getDist(const Vector3d& p) const {
99 
100  std::vector<double> a = { p.x-dx_, p.y-dx_, p.z-dx_ };
101  std::vector<double> b = { p.x+dx_, p.y+dx_, p.z+dx_ };
102  aabb::AABB box = aabb::AABB(a,b);
103  double mdist = 1e100; // far far away
104  auto indices = tree.query(box);
105  // std::cout << indices.size() << " segments in range\n";
106  for (int i : indices) {
107 
108  Vector3d x1 = nodes_[segments_[i].x];
109  Vector3d x2 = nodes_[segments_[i].y];
110  Vector3d v = x2.minus(x1);
111  Vector3d w = p.minus(x1);
112 
113  double c1 = v.times(w);
114  double c2 = v.times(v);
115 
116  double l;
117  if (c1<=0) {
118  l = w.length();
119  } else if (c1>=c2) {
120  l = p.minus(x2).length();
121  } else {
122  l = p.minus(x1.plus(v.times(c1/c2))).length();
123  }
124  l -= radii_[i];
125 
126  if (l < mdist) {
127  mdist = l;
128  }
129 
130  }
131  return -mdist;
132 }
