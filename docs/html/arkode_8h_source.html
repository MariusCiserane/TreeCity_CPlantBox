43 /*---------------------------------------------------------------
44  Enumerations for inputs to *StepCreate and *StepEvolve.
45  -----------------------------------------------------------------
46  itask: The itask input parameter to ARKode indicates the job
47  of the solver for the next user step. The ARK_NORMAL
48  itask is to have the solver take internal steps until
49  it has reached or just passed the user specified tout
50  parameter. The solver then interpolates in order to
51  return an approximate value of y(tout). The ARK_ONE_STEP
52  option tells the solver to just take one internal step
53  and return the solution at the point reached by that
54  step. If fixed time steps are desired, then the user
55  can run in ARK_ONE_STEP mode, and set hmin and hmax to
56  be the desired time step -- this will effectively
57  disable all temporal error control, and the user may
58  need to increase the iteration counts for the nonlinear
59  and linear solvers to guarantee convergence for larger
60  step sizes.
61  ---------------------------------------------------------------*/
62 
63 /* itask */
64 #define ARK_NORMAL 1
65 #define ARK_ONE_STEP 2

121 /*-----------------------------------------------------------------
122  Type : ARKRhsFn
123  -----------------------------------------------------------------
124  The f functions which define the right hand side of ODE
125  systems, f(t,y), must have type ARKRhsFn. f takes as input the
126  independent variable value t, and the dependent variable vector
127  y. It stores the result of f(t,y) in the vector ydot. The y
128  and ydot arguments are of type N_Vector.
129  (Allocation of memory for ydot is handled within ARKode)
130  The user_data parameter is the same as the user_data
131  parameter set by the user through the *StepSetUserData routine.
132  This user-supplied pointer is passed to the user's f()
133  function every time it is called.
134 
135  An ARKRhsFn should return 0 if successful, a negative value if
136  an unrecoverable error occured, and a positive value if a
137  recoverable error (e.g. invalid y values) occured.
138  If an unrecoverable occured, the integration is halted.
139  If a recoverable error occured, then (in most cases) ARKode
140  will try to correct and retry.
141  ---------------------------------------------------------------*/
142 typedef int (*ARKRhsFn)(realtype t, N_Vector y,
143  N_Vector ydot, void *user_data);
144 
145 /*-----------------------------------------------------------------
146  Type : ARKRootFn
147  -----------------------------------------------------------------
148  A function g, which defines a set of functions g_i(t,y) whose
149  roots are sought during the integration, must have type
150  ARKRootFn. The function g takes as input the independent
151  variable value t, and the dependent variable vector y. It
152  stores the nrtfn values g_i(t,y) in the realtype array gout.
153  (Allocation of memory for gout is handled within ARKode.)
154  The user_data parameter is the same as that passed by the user
155  to the *StepSetUserData routine. This user-supplied pointer
156  is passed to the user's g function every time it is called.
157 
158  An ARKRootFn should return 0 if successful or a non-zero value
159  if an error occured (in which case the integration will be
160  halted).
161  ---------------------------------------------------------------*/
162 typedef int (*ARKRootFn)(realtype t, N_Vector y,
163  realtype *gout, void *user_data);
164 
165 /*-----------------------------------------------------------------
166  Type : ARKEwtFn
167  -----------------------------------------------------------------
168  A function e, which sets the error weight vector ewt, must have
169  type ARKEwtFn. The function e takes as input the current
170  dependent variable y. It must set the vector of error weights
171  used in the WRMS norm:
172 
173  ||y||_WRMS = sqrt [ 1/N * sum ( ewt_i * y_i)^2 ]
174 
175  Typically, the vector ewt has components:
176 
177  ewt_i = 1 / (reltol * |y_i| + abstol_i)
178 
179  The user_data parameter is the same as that passed by the user
180  to the *StepSetUserData routine. This user-supplied pointer
181  is passed to the user's e function every time it is called.
182  An ARKEwtFn e must return 0 if the error weight vector has been
183  successfuly set and a non-zero value otherwise.
184  ---------------------------------------------------------------*/
185 typedef int (*ARKEwtFn)(N_Vector y, N_Vector ewt, void *user_data);
186 
187 /*-----------------------------------------------------------------
188  Type : ARKRwtFn
189  -----------------------------------------------------------------
190  A function r, which sets the residual weight vector rwt, must
191  have type ARKRwtFn. The function r takes as input the current
192  dependent variable y. It must set the vector of residual
193  weights used in the WRMS norm:
194 
195  ||v||_WRMS = sqrt [ 1/N * sum ( rwt_i * v_i)^2 ]
196 
197  Typically, the vector rwt has components:
198 
199  rwt_i = 1 / (reltol * |(M*y)_i| + rabstol_i)
200 
201  The user_data parameter is the same as that passed by the user
202  to the *StepSetUserData routine. This user-supplied pointer
203  is passed to the user's r function every time it is called.
204  An ARKRwtFn e must return 0 if the residual weight vector has
205  been successfuly set and a non-zero value otherwise.
206  ---------------------------------------------------------------*/
207 typedef int (*ARKRwtFn)(N_Vector y, N_Vector rwt, void *user_data);
208 
209 /*-----------------------------------------------------------------
210  Type : ARKErrHandlerFn
211  -----------------------------------------------------------------
212  A function eh, which handles error messages, must have type
213  ARKErrHandlerFn. The function eh takes as input the error code,
214  the name of the module reporting the error, the error message,
215  and a pointer to user data, the same as that passed to
216  *StepSetUserData.
217 
218  All error codes are negative, except ARK_WARNING which indicates

220 
221  An ARKErrHandlerFn has no return value.
222  ---------------------------------------------------------------*/
223 typedef void (*ARKErrHandlerFn)(int error_code, const char *module,
224  const char *function, char *msg,
225  void *user_data);

227 /*-----------------------------------------------------------------
228  Type : ARKAdaptFn
229  -----------------------------------------------------------------
230  A function which sets the new time step h, must have type
231  ARKAdaptFn. The function takes as input the current dependent
232  variable y, the current time t, the last 3 step sizes h,
233  the last 3 error estimates, the method order q, the embedding
234  order p, and a pointer to user data. The function must set the
235  scalar step size for the upcoming time step. This value will
236  subsequently be bounded by the user-supplied values for the
237  minimum and maximum allowed time step, and the time step
238  satisfying the explicit stability restriction. The user_data
239  parameter is the same as that passed by the user to the
240  *StepSetUserData routine. This user-supplied pointer is
241  passed to the function every time it is called.
242 
243  An ARKAdaptFn must return 0 if the new time step has been
244  successfuly set and a non-zero value otherwise.
245  ---------------------------------------------------------------*/
246 typedef int (*ARKAdaptFn)(N_Vector y, realtype t, realtype h1,
247  realtype h2, realtype h3,
248  realtype e1, realtype e2,
249  realtype e3, int q, int p,
250  realtype *hnew, void *user_data);
251 
252 /*-----------------------------------------------------------------
253  Type : ARKExpStabFn
254  -----------------------------------------------------------------
255  A function which returns the time step satisfying the stability
256  restriction for an explicit portion of the ODE. The function
257  takes as input the current dependent variable y, the current
258  time t, and a pointer to user data. The function must set the
259  scalar step size satisfying the stability restriction for the
260  upcoming time step. This value will subsequently be bounded by
261  the user-supplied values for the minimum and maximum allowed
262  time step, and the accuracy-based time step. The user_data
263  parameter is the same as that passed by the user to the
264  *StepSetUserData routine. This user-supplied pointer is passed
265  to the function every time it is called.
266 
267  If this function is not supplied (NULL), or if it returns a
268  negative time step size, then ARKode will assume that there is
269  no explicit stability restriction on the time step size.
270 
271  An ARKExpStabFn must return 0 if the step size limit has been
272  successfuly set and a non-zero value otherwise.
273  ---------------------------------------------------------------*/
274 typedef int (*ARKExpStabFn)(N_Vector y, realtype t,
275  realtype *hstab, void *user_data);

277 /*-----------------------------------------------------------------
278  Type : ARKVecResizeFn
279  -----------------------------------------------------------------
280  When calling *StepResize, the user may specify a vector resize
281  function to be used to convert any existing N_Vectors in the
282  ARKode memory structure to the new problem size. This would
283  typically be used if there is a user-supplied N_Vector module
284  that allows dynamic resizing of the vector data structures
285  without the need to delete/allocate memory on each call.
286 
287  The default behavior will be to delete the vector memory and
288  re-clone from the new vector; if this is the desired behavior
289  then specification of the ARKVecResizeFn is not recommended.
290 
291  The first argument, 'y', is the vector to be resized.
292 
293  The second argument, 'ytemplate', is the user-provided vector
294  with the "new" size, that may be used as a template.
295 
296  The third argument, 'user_data', is a user-provided data
297  structure to *StepResize, in case additional data is
298  necessary for the resize operation.
299 
300  An ARKVecResizeFn should return 0 if successful, and a nonzero
301  value if an error occurred.
302  ---------------------------------------------------------------*/
303 typedef int (*ARKVecResizeFn)(N_Vector y, N_Vector ytemplate,
304  void *user_data);

306 /*-----------------------------------------------------------------
307  Type : ARKPostProcessStepFn
308  -----------------------------------------------------------------
309  A function that is used to process the results of each timestep
310  solution, in preparation for subsequent steps. A routine of
311  this type is designed for tasks such as inter-processor
312  communication, computation of derived quantities, etc..
313 
314  IF THIS IS USED TO MODIFY ANY OF THE ACTIVE STATE DATA, THEN ALL
315  THEORETICAL GUARANTEES OF SOLUTION ACCURACY AND STABILITY ARE
316  LOST.
317 
318  Inputs:
319  t current time of ARKode solution
320  y current ARKode solution N_Vector for processing
321  user_data the structure passed by the user to the
322  ARKodeSetUserData routine.
323  ---------------------------------------------------------------*/
324 typedef int (*ARKPostProcessStepFn)(realtype t, N_Vector y,
325  void *user_data);
